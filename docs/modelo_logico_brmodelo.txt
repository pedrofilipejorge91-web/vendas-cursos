MODELO LOGICO - COMPATIVEL COM BRMODELO
Projeto: vendas-cursos

Legenda:
- PK = chave primaria
- FK = chave estrangeira
- UK = chave unica


USERS(
  id PK,
  name,
  email UK,
  email_verified_at,
  password,
  tipo,
  termos_aceites_em,
  privacidade_aceite_em,
  remember_token,
  created_at,
  updated_at
)

  id PK,
  user_id FK -> USERS.id,

  rua,

  contacto,

  updated_at
)
